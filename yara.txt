# Правила PC Sentinel в стиле YARA.
#
# Файл читается рядом с exe, в корне проекта и в каталоге данных приложения
# (%LOCALAPPDATA%\PCSentinel). Дополнять можно без пересборки.
#
# Формат:
#   rule <Имя>
#     severity = critical | warning
#     text = "строка"        — поиск без учёта регистра
#     hex  = 4D 5A ?? 00     — точные байты, ?? — любой байт
#     condition = any | all  — достаточно одного совпадения или нужны все
#
# Правила с condition = all надёжнее: одно слово почти всегда встречается
# и в обычных программах.

# --- Скрипты и загрузчики ----------------------------------------------

rule EncodedPowerShell
  severity = warning
  text = "powershell"
  text = "-encodedcommand"
  condition = all

rule HiddenWindowLaunch
  severity = warning
  text = "-windowstyle hidden"
  text = "bypass"
  condition = all

  condition = all
